# 本地编码助手作为模型提供者方案

## 目标

在 Agent4Market 的模型设置中，除了现有的 NewAPI 网关外，新增 Claude Code 和 Codex CLI 作为可选的模型提供者。用户可以直接选择使用本地编码助手执行任务。

## 设计原则

1. **统一接口**：编码助手以 subprocess 方式调用，封装为标准的模型提供者接口
2. **自动发现**：检测本地是否安装 Claude Code 和 Codex CLI
3. **无缝切换**：用户可以在不同模型提供者之间切换
4. **审批一致**：使用编码助手作为模型时，审批边界保持不变

```python
def detect_coding_assistants(
    environ: Mapping[str, str] | None = None

    environment = dict(os.environ if environ is None else environ)
```

#### 3.2 编码助手模型执行器

```python
class CodingAssistantModelProvider:
    """使用编码助手作为模型提供者"""

    def __init__(
        self,
        provider_type: str,  # "claude-code" 或 "codex-cli"
        executable_path: str,
        model_id: str,
        project_root: Path,
    ):
        self.provider_type = provider_type
        self.executable_path = executable_path
        self.model_id = model_id
        self.project_root = project_root

    def execute_task(
        self,
        service_id: str,
        project_id: str,
        request_text: str,
        thinking_level: str = "high",
    ) -> dict[str, Any]:
        """通过编码助手执行任务"""
        # 构建命令
        if self.provider_type == "claude-code":
            cmd = [
                self.executable_path,
                "--model", self.model_id,
                "--thinking", thinking_level,
                "/agent4market-sales-director",
                f"创建任务：服务={service_id}，项目={project_id}，请求={request_text}"
            ]
        else:  # codex-cli
            cmd = [
                self.executable_path,
                "chat",
                "--model", self.model_id,
                f"使用 /agent4market-sales-director skill，创建任务：服务={service_id}，项目={project_id}，请求={request_text}"
            ]

        # 执行命令
        result = subprocess.run(
            cmd,
            cwd=str(self.project_root),
            capture_output=True,
            text=True,
            timeout=300,
        )

        if result.returncode != 0:
            raise RuntimeError(f"编码助手执行失败：{result.stderr}")

        # 解析输出，提取任务 ID
        return self._parse_output(result.stdout)

    def _parse_output(self, output: str) -> dict[str, Any]:
        """从编码助手输出中提取任务信息"""
        # 查找 request-xxxxx 格式的任务 ID
        import re
        match = re.search(r'request-[a-f0-9]{12}', output)
        if match:
            return {
                "status": "ok",
                "task_id": match.group(0),
                "output": output,
            }
        else:
            return {
                "status": "unknown",
                "output": output,
            }
```

#### 3.3 统一模型接口

```python
def get_model_executor(
    project_root: Path | str
) -> CodingAssistantModelProvider | NewAPIModelProvider:
    """根据配置返回对应的模型执行器"""
    settings = load_model_settings(project_root)
    if not settings:
        raise ModelProviderError("未配置模型提供者")

    provider_type = settings.get("provider_type", "newapi")

    if provider_type == "newapi":
        # 现有的 NewAPI 逻辑
        return NewAPIModelProvider(...)

    elif provider_type == "claude-code":
        return CodingAssistantModelProvider(
            provider_type="claude-code",
            executable_path=settings["claude_code_path"],
            model_id=settings["claude_code_model"],
            project_root=Path(project_root),
        )

    elif provider_type == "codex-cli":
        return CodingAssistantModelProvider(
            provider_type="codex-cli",
            executable_path=settings["codex_cli_path"],
            model_id=settings["codex_cli_model"],
            project_root=Path(project_root),
        )

    else:
        raise ModelProviderError(f"未知的模型提供者类型：{provider_type}")
```

#### 4.3 JavaScript 逻辑

```javascript
// 检测编码助手
async function detectCodingAssistants() {
  try {
    const response = await fetch("/api/coding-assistants/detect");
    const data = await response.json();

    // 更新 Claude Code 状态
    const claudeStatus = $("claude-code-status");
    const claudeCard = document.querySelector('input[value="claude-code"]').closest('.provider-card');
    if (data["claude-code"]?.available) {
      claudeStatus.textContent = `已安装 (${data["claude-code"].version})`;
      claudeStatus.className = "provider-status available";
      claudeCard.querySelector('input').disabled = false;
      $("claude-code-path").value = data["claude-code"].path;
    } else {
      claudeStatus.textContent = "未检测到";
      claudeStatus.className = "provider-status unavailable";
      claudeCard.querySelector('input').disabled = true;
    }

    // 更新 Codex CLI 状态
    const codexStatus = $("codex-cli-status");
    const codexCard = document.querySelector('input[value="codex-cli"]').closest('.provider-card');
    if (data["codex-cli"]?.available) {
      codexStatus.textContent = `已安装 (${data["codex-cli"].version})`;
      codexStatus.className = "provider-status available";
      codexCard.querySelector('input').disabled = false;
      $("codex-cli-path").value = data["codex-cli"].path;
    } else {
      codexStatus.textContent = "未检测到";
      codexStatus.className = "provider-status unavailable";
      codexCard.querySelector('input').disabled = true;
    }
  } catch (error) {
    note("检测编码助手失败：" + error.message, true);
  }
}

// 提供者类型切换
document.querySelectorAll('input[name="provider-type"]').forEach(radio => {
  radio.addEventListener('change', (e) => {
    const selected = e.target.value;

    // 隐藏所有配置面板
    $("claude-code-config").hidden = true;
    $("codex-cli-config").hidden = true;
    $("model-settings-panel").hidden = true;

    // 显示对应的配置面板
    if (selected === "newapi") {
      $("model-settings-panel").hidden = false;
    } else if (selected === "claude-code") {
      $("claude-code-config").hidden = false;
    } else if (selected === "codex-cli") {
      $("codex-cli-config").hidden = false;
    }

    $("provider-current").textContent = e.target.closest('.provider-card').querySelector('strong').textContent;
  });
});

// 保存提供者设置
$("save-provider-settings")?.addEventListener("click", async () => {
  const selectedProvider = document.querySelector('input[name="provider-type"]:checked')?.value;

  if (!selectedProvider) {
    note("请选择一个模型提供者", true);
    return;
  }

  const payload = {
    provider_type: selectedProvider,
  };

  if (selectedProvider === "claude-code") {
    payload.claude_code_path = $("claude-code-path").value;
    payload.claude_code_model = $("claude-code-model").value;
  } else if (selectedProvider === "codex-cli") {
    payload.codex_cli_path = $("codex-cli-path").value;
    payload.codex_cli_model = $("codex-cli-model").value;
  }

  try {
    const response = await fetch("/api/model-provider/configure", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });

    const data = await response.json();
    if (data.status === "ok") {
      note("模型提供者已保存，请重启应用生效。");
    } else {
      note("保存失败：" + data.error, true);
    }
  } catch (error) {
    note("网络错误：" + error.message, true);
  }
});

// 重新检测
$("detect-coding-assistants")?.addEventListener("click", detectCodingAssistants);

// 页面加载时检测
if (currentView === "settings") {
```

## 优势

1. **灵活性**：用户可根据需求选择最合适的模型提供者
2. **本地优先**：可以完全使用本地编码助手，无需外部网关
3. **统一体验**：无论使用哪种提供者，审批流程和界面保持一致
4. **简化配置**：自动检测本地编码助手，无需手动配置路径

## 局限性

1. **执行方式不同**：编码助手通过 subprocess 调用，可能比直接 API 调用慢
2. **输出解析**：需要从编码助手的文本输出中提取结构化信息
3. **错误处理**：编码助手的错误信息可能不如 API 结构化
