## JavaScript 改动

在 `ui/app.js` 文件的 `switchView("settings")` 相关逻辑中添加以下代码：

### 1. 检测编码助手函数

在文件末尾添加：

```javascript
// 检测本地编码助手
async function detectCodingAssistants() {
  try {
    const response = await fetch("/api/coding-assistants/detect");
    const data = await response.json();

    // 更新 Claude Code 状态
    const claudeStatus = $("claude-code-status");
    const claudeRadio = document.querySelector('input[value="claude-code"]');
    if (data["claude-code"]?.available) {
      claudeStatus.textContent = `已安装 (${data["claude-code"].version})`;
      claudeStatus.className = "provider-status available";
      claudeRadio.disabled = false;
      $("claude-code-path").value = data["claude-code"].path;
    } else {
      claudeStatus.textContent = "未检测到";
      claudeStatus.className = "provider-status unavailable";
      claudeRadio.disabled = true;
    }

    // 更新 Codex CLI 状态
    const codexStatus = $("codex-cli-status");
    const codexRadio = document.querySelector('input[value="codex-cli"]');
    if (data["codex-cli"]?.available) {
      codexStatus.textContent = `已安装 (${data["codex-cli"].version})`;
      codexStatus.className = "provider-status available";
      codexRadio.disabled = false;
      $("codex-cli-path").value = data["codex-cli"].path;
    } else {
      codexStatus.textContent = "未检测到";
      codexStatus.className = "provider-status unavailable";
      codexRadio.disabled = true;
    }
  } catch (error) {
    note("检测编码助手失败：" + error.message, true);
  }
}

// 提供者类型切换
document.querySelectorAll('input[name="provider-type"]')?.forEach(radio => {
  radio.addEventListener('change', (e) => {
    const selected = e.target.value;

    // 隐藏所有配置面板
    $("claude-code-config").hidden = true;
    $("codex-cli-config").hidden = true;

    // 显示对应的配置面板
    if (selected === "claude-code") {
      $("claude-code-config").hidden = false;
    } else if (selected === "codex-cli") {
      $("codex-cli-config").hidden = false;
    }

    $("provider-current").textContent = e.target.closest('.provider-card').querySelector('strong').textContent;
  });
});

// 保存提供者设置
$("save-provider-settings")?.addEventListener("click", async () => {
  const selectedRadio = document.querySelector('input[name="provider-type"]:checked');
  if (!selectedRadio) {
    note("请选择一个模型提供者", true);
    return;
  }

  const selectedProvider = selectedRadio.value;
  const btn = $("save-provider-settings");
  btn.disabled = true;
  btn.textContent = "保存中…";

  try {
    let payload = { provider_type: selectedProvider };

    if (selectedProvider === "newapi") {
      // NewAPI 使用现有的配置逻辑
      note("请在下方 NewAPI 配置面板中配置网关和模型", false);
      btn.disabled = false;
      btn.textContent = "保存提供者选择";
      return;
    } else if (selectedProvider === "claude-code") {
      payload.executable_path = $("claude-code-path").value;
      payload.selected_model = $("claude-code-model").value;
    } else if (selectedProvider === "codex-cli") {
      payload.executable_path = $("codex-cli-path").value;
      payload.selected_model = $("codex-cli-model").value;
    }

    const response = await fetch("/api/model-provider/configure", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });

    const data = await response.json();
    if (data.status === "ok") {
      note(data.message || "模型提供者已保存，请重启应用生效。");
      $("provider-current").textContent = selectedRadio.closest('.provider-card').querySelector('strong').textContent;
    } else {
      note("保存失败：" + (data.error || "未知错误"), true);
    }
  } catch (error) {
    note("网络错误：" + error.message, true);
  } finally {
    btn.disabled = false;
    btn.textContent = "保存提供者选择";
  }
});

// 重新检测按钮
$("detect-coding-assistants")?.addEventListener("click", detectCodingAssistants);
```

## 测试步骤

1. 启动工作台
2. 进入"设置"页面
3. 查看"模型提供者"面板
4. 确认 Claude Code 和 Codex CLI 的检测状态
5. 选择一个可用的提供者
6. 选择对应的模型
7. 点击"保存提供者选择"
8. 重启应用
9. 创建任务验证新的模型提供者是否生效

## 注意事项

- Claude Code 和 Codex CLI 必须已安装并在 PATH 中可访问
- 切换提供者后需要重启应用才能生效
- 使用本地编码助手时，会使用系统环境中的 Anthropic API 密钥
- NewAPI 配置面板保持原样，当选择 NewAPI 时使用现有配置
